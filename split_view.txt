+find_first_editor_recursive(GtkWidget *widget) {
+    if (!widget) return NULL;
+    if (EDITOR_IS_WIDGET(widget)) return widget;
+

+static void
+close_split_view(GtkWidget *overlay)
+{
+    if (!overlay || !GTK_IS_OVERLAY(overlay)) return;
+    
+    GtkWidget *parent = gtk_widget_get_parent(overlay);
+    if (!parent) return;
+    
+    /* Case 1: Parent is TabRoot (Box) -> Closing the last view */
+    if (GTK_IS_BOX(parent)) {
+        /* Find the tab associated with this view to close it */
+        GtkRoot *root = gtk_widget_get_root(overlay);
+        ViteWindow *win = g_object_get_data(G_OBJECT(root), "vite-window");
+        
+        /* Ideally we find the specific tab. 
+           We know `close_split_view` is called from a button on the overlay. 
+           And the overlay is in a tab. which tab?
+           We can iterate tabs to find which one has `page_wrapper == parent` (if parent is TabRoot)?
+           Actually, simpler: 
+           `on_tab_close_clicked` takes a ViteTab.
+           We can find the ViteTab by walking up or checking `last_focused_child` logic?
+           
+           Wait, `do_split` relies on `vite_tab_bar_get_active_tab`.
+           If we click close on a background tab's split view, we should close THAT tab/view.
+           
+           We stored "vite-window" on the overlay.
+           How to get the Tab from the Overlay?
+           We iterate the tabs in the window.
+           For each tab, get "page".
+           Check if `page` is an ancestor of `overlay`.
+        */
+        if (win) {
+            GList *tabs = vite_tab_bar_get_tabs(win->tab_bar);
+            for (GList *l = tabs; l != NULL; l = l->next) {
+                ViteTab *t = VITE_TAB(l->data);
+                GtkWidget *page = g_object_get_data(G_OBJECT(t), "page");
+                /* If parent is box, then parent IS the page root (TabRoot) */
+                if (page == parent) {
+                    on_tab_close_clicked(t, NULL);
+                    break;
+                }
+            }
+            g_list_free(tabs);
+        }
+        return;
+    }
+    
+    /* Case 2: Parent is Paned -> Collapsing a split */
+    if (GTK_IS_PANED(parent)) {
+        GtkPaned *paned = GTK_PANED(parent);
+        GtkWidget *start = gtk_paned_get_start_child(paned);
+        GtkWidget *end = gtk_paned_get_end_child(paned);
+        
+        GtkWidget *sibling = (start == overlay) ? end : start;
+        
+        /* If sibling is NULL (shouldn't happen in valid split), just destroy parent? */
+        if (!sibling) {
+             /* Weird state, just remove myself */
+             if (start == overlay) gtk_paned_set_start_child(paned, NULL);
+             else gtk_paned_set_end_child(paned, NULL);
+             return;
+        }
+        
+        /* Promote sibling to replace parent */
+        GtkWidget *grandparent = gtk_widget_get_parent(parent);
+        if (!grandparent) return;
+        
+        /* Fix: Clear focus globally to prevent GtkPaned warnings about lost focus child */
+        GtkRoot *root = gtk_widget_get_root(overlay);
+        ViteWindow *win = g_object_get_data(G_OBJECT(root), "vite-window");
+        if (win && win->window) {
+             /* Only clear if focus is currently inside the hierarchy we are touching? 
+                Safer to just clear. */
+             gtk_window_set_focus(win->window, NULL);
+        }
+        
+        g_object_ref(sibling); /* Protect sibling during reparent */
+        
+        /* Detach sibling from paned first? 
+           GTK4: setting child to NULL removes it.
+           We need to remove it to reparent it. */
+        if (sibling == start) gtk_paned_set_start_child(paned, NULL);
+        else gtk_paned_set_end_child(paned, NULL);
+        
+        /* Remove overlay (target) to clean up */
+        if (overlay == start) gtk_paned_set_start_child(paned, NULL);
+        else gtk_paned_set_end_child(paned, NULL);
+        
+        /* Now replace parent with sibling in grandparent */
+        if (GTK_IS_BOX(grandparent)) {
+            /* TabRoot */
+            /* Remove paned, append sibling. 
+               Box has only 1 child? Yes. */
+            gtk_box_remove(GTK_BOX(grandparent), parent);
+            gtk_box_append(GTK_BOX(grandparent), sibling);
+        } else if (GTK_IS_PANED(grandparent)) {
+            GtkPaned *gp = GTK_PANED(grandparent);
+            if (gtk_paned_get_start_child(gp) == parent) {
+                gtk_paned_set_start_child(gp, NULL); /* Remove old paned */
+                gtk_paned_set_start_child(gp, sibling);
+            } else {
+                gtk_paned_set_end_child(gp, NULL);
+                gtk_paned_set_end_child(gp, sibling);
+            }
+        }
+        
+        g_object_unref(sibling); /* Grandparent holds it now */
+        
+        /* Restore focus to sibling */
+        GtkWidget *new_focus = find_first_editor_recursive(sibling);
+        if (new_focus) {
+            defer_focus(new_focus);
+            if (GTK_IS_OVERLAY(sibling)) {
+                 GtkWidget *s_btn = g_object_get_data(G_OBJECT(sibling), "close-btn");
+                 if (s_btn) {
+                     GtkWidget *s_parent = gtk_widget_get_parent(sibling);
+                     gtk_widget_set_visible(s_btn, GTK_IS_PANED(s_parent));
+                 }
+            }
+        }
+    }

+static void
+do_split(ViteWindow *win, GtkOrientation orientation)
+{
+    /* 1. Get Active Tab */
+    ViteTab *tab = vite_tab_bar_get_active_tab(win->tab_bar);
+    if (!tab) return;
+    
+    GtkWidget *page = g_object_get_data(G_OBJECT(tab), "page");
+    if (!page) return;
+    
+    /* 2. Try to use tracking first */
+    GtkWidget *target_overlay = vite_tab_get_last_focused_child(tab);
+    
+    /* Validate tracked overlay: must be inside the current page */
+    if (target_overlay) {
+        if (!gtk_widget_is_ancestor(target_overlay, page) && target_overlay != page) {
+             target_overlay = NULL;
+        }
+    }
+    
+    /* Fallback: Window Focus */
+    if (!target_overlay) {
+        GtkWidget *focus = gtk_window_get_focus(win->window);
+        if (focus && (gtk_widget_is_ancestor(focus, page) || focus == page)) {
+             /* Walk up to overlay */
+             target_overlay = focus;
+             while (target_overlay && !GTK_IS_OVERLAY(target_overlay) && target_overlay != page) {
+                 target_overlay = gtk_widget_get_parent(target_overlay);
+             }
+        }
+    }
+
+    /* Fallback: First editor */
+    if (!target_overlay || !GTK_IS_OVERLAY(target_overlay)) {
+        GtkWidget *ed = find_first_editor_recursive(page);
+        if (ed) {
+             target_overlay = gtk_widget_get_ancestor(ed, GTK_TYPE_OVERLAY);
+        }
+    }
+    
+    if (!target_overlay || !GTK_IS_OVERLAY(target_overlay)) return;
+
+    /* 4. Get Parent of Overlay */
+    GtkWidget *parent = gtk_widget_get_parent(target_overlay);
+    if (!parent) return;
+
+    /* 5. Create Paned */
+    GtkWidget *paned = gtk_paned_new(orientation);
+    
+    /* 6. Remove target from parent and replace with paned */
+    g_object_ref(target_overlay); /* Prevent destruction */
+    
+    if (GTK_IS_BOX(parent)) {
+        gtk_box_remove(GTK_BOX(parent), target_overlay);
+        gtk_box_append(GTK_BOX(parent), paned);
+        /* If TabRoot Box had multiple children (shouldn't happen with current logic unless we allow side-by-side without paned?), 
+           append adds to end. Assuming TabRoot has only 1 child initially or we are replacing the specific one.
+           Wait, GtkBox remove/append works for single child scenario. 
+           If parent has other children, appending might put it at global end.
+           Since TabRoot (Box) will only hold the root of the split tree, 
+           initially it holds 1 Overlay. After split, it holds 1 Paned.
+           So remove/append is effectively replace. */
+    } else if (GTK_IS_PANED(parent)) {
+        if (gtk_paned_get_start_child(GTK_PANED(parent)) == target_overlay) {
+             gtk_paned_set_start_child(GTK_PANED(parent), NULL);
+             gtk_paned_set_start_child(GTK_PANED(parent), paned);
+        } else {
+             gtk_paned_set_end_child(GTK_PANED(parent), NULL);
+             gtk_paned_set_end_child(GTK_PANED(parent), paned);
+        }
+    } else {
+        g_object_unref(target_overlay);
+        g_warning("Unknown parent type for split");
+        return;
+    }
+    
+    /* 7. Put target in start of new paned */
+    gtk_paned_set_start_child(GTK_PANED(paned), target_overlay);
+    g_object_unref(target_overlay); /* Release our ref, Paned holds it now */
+    
+    /* 8. Create clone for end of new paned */
+    GtkWidget *old_editor = find_first_editor_recursive(target_overlay);
+    Document *doc = editor_widget_get_document(EDITOR_WIDGET(old_editor));
+    
+    GtkWidget *new_scrolled = gtk_scrolled_window_new();
+    GtkWidget *new_editor = editor_widget_new();
+    gtk_scrolled_window_set_child(GTK_SCROLLED_WINDOW(new_scrolled), new_editor);
+    editor_widget_set_document(EDITOR_WIDGET(new_editor), doc);
+    
+    const char *path = document_get_file_path(doc);
+    if (path) {
+         const char *dot = strrchr(path, '.');
+         if (dot) editor_widget_set_language(EDITOR_WIDGET(new_editor), dot + 1);
+    }
+    
+    /* Replicate generic connection logic? 
+       Actually Document logic handles content/modified. 
+       But we might want to ensure 'tab' is not lost? 
+       The overlay in create_view_container sets 'vite-window'. 
+       We need to create the wrapper. */
+    
+    GtkWidget *new_overlay = create_view_container(win, new_scrolled);
+    
+    gtk_paned_set_end_child(GTK_PANED(paned), new_overlay);
+    
+    /* Set balanced position (50%) */
+    /* Needs allocation to know size, but we can try setting half of current target size */
+    int size = (orientation == GTK_ORIENTATION_HORIZONTAL) ? gtk_widget_get_width(target_overlay) : gtk_widget_get_height(target_overlay);
+    if (size > 0) gtk_paned_set_position(GTK_PANED(paned), size / 2);
+    
+    /* 9. Focus new editor */
+    defer_focus(new_editor);
+}
